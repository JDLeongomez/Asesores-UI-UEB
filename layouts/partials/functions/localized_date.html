"dic." }} {{ $monthIndex := sub (int ($date.Format "1")) 1 }} {{ $layout = replace $layout "January" (index $months $monthIndex) }} {{ $layout = replace $layout "Jan" (index $monthsShort $monthIndex) }} {{ $date.Format $layout }}
